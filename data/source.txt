 	.BEGIN 	
	IN 	VALUE
	IN 	POWER
	LOAD 	VALUE
	STORE 	SUM
		
L1: 	DECREMENT 	POWER
	LOAD 	POWER
	COMPARE 	ZERO
	JUMPEQ 	L3
	LOAD 	SUM
	STORE 	PART
	LOAD 	VALUE
	STORE 	COUNT
		
L2: 	DECREMENT 	COUNT
	LOAD 	COUNT
	COMPARE 	ZERO
	JUMPEQ 	L1
	LOAD 	SUM
	ADD 	PART
	STORE 	SUM
	JUMP L2 	
		
L3: 	OUT 	SUM
	HALT 	
		
ZERO: 	.DATA 	0
VALUE: 	.DATA 	0
POWER: 	.DATA 	0
COUNT: 	.DATA 	0
PART: 	.DATA 	0
SUM: 	.DATA 	0
	.END
